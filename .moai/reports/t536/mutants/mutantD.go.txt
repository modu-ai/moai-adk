package kanban

// MUTANT D (temporary, never compiled in place): the init-SNAPSHOT shape — the
// root set is copied into a package variable at initialization instead of being
// read through the seam at call time, so every later stub is ignored silently.

import (
	"os"
	"path/filepath"
	"strings"
)

var TempRootsFn = defaultTempRoots

// The mutation: the set is frozen here, before any test can stub the seam.
var tempRootsSnapshot = TempRootsFn()

func defaultTempRoots() []string {
	return []string{os.TempDir(), "/tmp", "/var/folders"}
}

func TempOriginReason(base string) (reason string, isTemp bool) {
	if strings.TrimSpace(base) == "" {
		return "", false
	}
	abs, err := filepath.Abs(base)
	if err != nil {
		return "", false
	}
	normalizedBase := normalizeForTempCompare(abs)
	for _, root := range tempRootsSnapshot {
		if strings.TrimSpace(root) == "" {
			continue
		}
		rootAbs, err := filepath.Abs(root)
		if err != nil {
			continue
		}
		for _, anchor := range []string{normalizeForTempCompare(rootAbs), filepath.Clean(rootAbs)} {
			if pathWithin(normalizedBase, anchor) {
				return root, true
			}
		}
	}
	return "", false
}

func normalizeForTempCompare(path string) string {
	if _, err := os.Lstat(path); err != nil {
		return filepath.Clean(path)
	}
	if resolved, err := filepath.EvalSymlinks(path); err == nil {
		return resolved
	}
	return filepath.Clean(path)
}

func pathWithin(p, dir string) bool {
	rel, err := filepath.Rel(dir, p)
	if err != nil {
		return false
	}
	if rel == "." {
		return true
	}
	return rel != ".." && !strings.HasPrefix(rel, ".."+string(filepath.Separator))
}
