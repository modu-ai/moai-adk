package cli

import (
	"strings"
	"testing"

	"github.com/modu-ai/moai-adk/internal/profile"
)

func zzEffortFlags(args []string) int {
	n := 0
	for _, a := range args {
		if a == "--effort" || strings.HasPrefix(a, "--effort=") {
			n++
		}
	}
	return n
}

// Probe: an operator --effort anywhere in argv (before or after `--`,
// space or = form) must suppress the profile's --effort max injection.
func TestZZF2ProbeOperatorEffortAnywhere(t *testing.T) {
	root := withCrossSessionConfig(t, "")
	withLaunchEffortPrefs(t, profile.ProfilePreferences{EffortLevel: "max"}, nil)
	for _, op := range [][]string{
		{"--effort", "low"},
		{"--effort=low"},
		{"--", "--effort", "low"},
		{"--", "--effort=low"},
	} {
		got := appendCrossSessionSettings(root, "dev", op)
		if n := zzEffortFlags(got); n != 1 {
			t.Errorf("general op=%v argv=%v effortFlags=%d want 1", op, got, n)
		}
		flag, cleanup := prepareKanbanSettings("dev", op)
		cleanup()
		if n := zzEffortFlags(flag); n != 0 {
			t.Errorf("kanban op=%v injected=%v effortFlags=%d want 0", op, flag, n)
		}
	}
}
