// PROBE (throwaway, card t586): dump the huh v1 surfaces exactly as init.go /
// update.go / profile_setup.go construct them, to reproduce the F10 English
// fixed strings and the F11 layout on the v1 theme.
var t586V1ANSI = regexp.MustCompile("\x1b\\[[0-9;?]*[ -/]*[@-~]")

func t586V1Dump(t *testing.T, label string, f *huh.Form) {
	t.Helper()
	f.Init()
	m, _ := f.Update(tea.WindowSizeMsg{Width: 80, Height: 40})
	plain := t586V1ANSI.ReplaceAllString(m.(*huh.Form).View(), "")
	for i, ln := range strings.Split(plain, "\n") {
		t.Logf("PROBE %s %02d|%s|", label, i, strings.ReplaceAll(ln, " ", "·"))
	}
}

// TestProbeT586V1TTY runs the init/update profile confirm on a real pty for a
// tmux capture. Skipped unless the t586 pty harness sets MOAI_T586_TTY=1.
func TestProbeT586V1TTY(t *testing.T) {
	if os.Getenv("MOAI_T586_TTY") != "1" {
		t.Skip("tty probe runs only inside the t586 pty harness")
	}
	var wantSetup bool
	confirm := huh.NewConfirm().
		Title("No profile found. Set up profile preferences now?").
		Description("Configure your name, language, and model preferences.").
		Value(&wantSetup)
	_ = huh.NewForm(huh.NewGroup(confirm)).WithTheme(moaiHuhTheme()).Run()
}

func TestProbeT586HuhV1Surfaces(t *testing.T) {
	var wantSetup bool
	confirm := huh.NewConfirm().
		Title("No profile found. Set up profile preferences now?").
		Description("Configure your name, language, and model preferences.").
		Value(&wantSetup)
	t586V1Dump(t, "profile-confirm", huh.NewForm(huh.NewGroup(confirm)).WithTheme(moaiHuhTheme()))

	var convLang string
	langForm := huh.NewForm(
		huh.NewGroup(
			huh.NewSelect[string]().
				Title("Select your language").
				Description("Language for this wizard and Claude's responses.").
				Options(
					huh.NewOption("English", "en"),
					huh.NewOption("Korean (한국어)", "ko"),
					huh.NewOption("Japanese (日本語)", "ja"),
					huh.NewOption("Chinese (中文)", "zh"),
				).
				Value(&convLang),
		).Title("Language"),
	).WithTheme(moaiHuhTheme())
	t586V1Dump(t, "profile-lang-step", langForm)
}
