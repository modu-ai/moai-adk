// PROBE (throwaway, card t586): dump the rendered huh v2 wizard view with ANSI
// stripped and spaces made visible, to reproduce the F11 layout defects
// (centred confirm buttons, stacked blank lines, uneven option widths).
var t586ANSI = regexp.MustCompile("\x1b\\[[0-9;?]*[ -/]*[@-~]")

func t586Dump(t *testing.T, label, view string) {
	t.Helper()
	plain := t586ANSI.ReplaceAllString(view, "")
	for i, ln := range strings.Split(plain, "\n") {
		t.Logf("PROBE %s %02d|%s|", label, i, strings.ReplaceAll(ln, " ", "·"))
	}
}

func TestProbeT586WizardConfirmGroup(t *testing.T) {
	for _, loc := range []string{"en", "ko"} {
		locale := loc
		result := &WizardResult{}
		qa := &Question{ID: "probe_confirm_a", Type: QuestionTypeConfirm, Title: "Enable A?", Description: "First confirm description.", Default: "true"}
		qb := &Question{ID: "probe_confirm_b", Type: QuestionTypeConfirm, Title: "Enable B?", Description: "Second confirm description.", Default: "false"}
		qs := &Question{ID: "probe_select", Type: QuestionTypeSelect, Title: "Pick one", Description: "Select description.", Default: "a",
			Options: []Option{
				{Label: "Short", Value: "a", Desc: "x"},
				{Label: "Medium (Recommended)", Value: "b", Desc: "a much longer description string"},
			}}
		form := huh.NewForm(huh.NewGroup(
			buildConfirmField(qa, result, &locale),
			buildConfirmField(qb, result, &locale),
			buildSelectField(qs, result, &locale),
		)).WithTheme(newMoAIWizardTheme()).WithAccessible(false)
		d := newFormDriver(t, form)
		t586Dump(t, "confirm-group-"+locale, d.view())
	}
}

func TestProbeT586WizardFirstPage(t *testing.T) {
	result := &WizardResult{}
	form := buildUnifiedForm(InitQuestions(t.TempDir()), result, "ko")
	d := newFormDriver(t, form)
	t586Dump(t, "init-first-page-ko", d.view())
}
