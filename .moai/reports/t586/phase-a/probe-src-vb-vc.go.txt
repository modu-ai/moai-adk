var probeANSI = regexp.MustCompile(`\x1b\[[0-9;?]*[ -/]*[@-~]|\x1b\][^\x07\x1b]*(\x07|\x1b\\)`)

func probeDump(t *testing.T, label, view string) {
	t.Helper()
	for i, line := range strings.Split(probeANSI.ReplaceAllString(view, ""), "\n") {
		t.Logf("PROBE %s %02d|%s|", label, i, line)
	}
}

func probeKoConfirm(km *huh.KeyMap) *huh.Form {
	value := true
	ui := GetUIStrings("ko")
	conf := huh.NewConfirm().
		Title("다운그레이드 하시겠습니까?").
		Description("probe description").
		Affirmative(ui.ConfirmYes).
		Negative(ui.ConfirmNo).
		Value(&value)
	f := huh.NewForm(huh.NewGroup(conf)).WithTheme(newMoAIWizardTheme()).WithAccessible(false)
	if km != nil {
		f = f.WithKeyMap(km)
	}
	return f
}

// V-b: does a ko key map (NewDefaultKeyMap + SetHelp + WithKeyMap) change the
// help labels of a one-confirm form, including the y/n entries?
func TestProbeT586VbKeyMapConfirm(t *testing.T) {
	// control: no key map
	probeDump(t, "vb-no-keymap", newFormDriver(t, probeKoConfirm(nil)).view())

	// ko key map, y/n help set to distinct key-map-only strings so the source
	// of the rendered y/n label is attributable.
	km := huh.NewDefaultKeyMap()
	km.Confirm.Toggle.SetHelp("←/→", "전환")
	km.Confirm.Next.SetHelp("enter", "다음")
	km.Confirm.Submit.SetHelp("enter", "제출")
	km.Confirm.Prev.SetHelp("shift+tab", "이전")
	km.Confirm.Accept.SetHelp("y", "KM-ACCEPT")
	km.Confirm.Reject.SetHelp("n", "KM-REJECT")
	d := newFormDriver(t, probeKoConfirm(km))
	probeDump(t, "vb-ko-keymap-frame1", d.view())
	probeDump(t, "vb-ko-keymap-frame2", d.view())
}

// probeProfileResult is a non-*WizardResult binding target (V-c).
type probeProfileResult struct {
	First string
	Extra string
}

// V-c: a stepper whose visibility predicate is a closure over a
// non-*WizardResult struct recomputes when an answer changes.
func TestProbeT586VcGeneralizedStepper(t *testing.T) {
	res := &probeProfileResult{}
	ids := []string{"first", "extra", "last"}
	visible := func(id string) bool {
		if id == "extra" {
			return res.First == "yes"
		}
		return true
	}
	denom := func() int {
		n := 0
		for _, id := range ids {
			if visible(id) {
				n++
			}
		}
		return n
	}
	index := func(target string) int {
		n := 0
		for _, id := range ids {
			if !visible(id) {
				continue
			}
			n++
			if id == target {
				return n
			}
		}
		return n
	}
	note := huh.NewNote().TitleFunc(func() string {
		return tui.Stepper(index("first"), denom(), nil)
	}, res)

	first := "no"
	sel := huh.NewSelect[string]().Title("First?").
		Options(huh.NewOption("no", "no"), huh.NewOption("yes", "yes")).
		Value(&first)
	sel.Validate(func(v string) error { res.First = v; return nil })
	last := ""
	inp := huh.NewInput().Title("Last").Value(&last)

	form := huh.NewForm(huh.NewGroup(note, sel, inp)).WithTheme(newMoAIWizardTheme()).WithAccessible(false)
	d := newFormDriver(t, form)
	before := d.view()
	probeDump(t, "vc-before", before)
	d.down()  // highlight "yes"
	d.enter() // blur → Validate sets res.First = "yes"
	after := d.view()
	probeDump(t, "vc-after", after)
	t.Logf("PROBE vc res.First=%q", res.First)
}
