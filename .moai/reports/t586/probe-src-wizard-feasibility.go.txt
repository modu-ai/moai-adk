// PROBE (throwaway, card t586): check which F11 layout knobs are reachable from
// MoAI code without patching huh. Each variant changes one knob; the dump is
// compared against TestProbeT586WizardConfirmGroup.
func t586Questions() (*Question, *Question, *Question) {
	qa := &Question{ID: "probe_confirm_a", Type: QuestionTypeConfirm, Title: "Enable A?", Description: "First confirm description.", Default: "true"}
	qb := &Question{ID: "probe_confirm_b", Type: QuestionTypeConfirm, Title: "Enable B?", Description: "Second confirm description.", Default: "false"}
	qs := &Question{ID: "probe_select", Type: QuestionTypeSelect, Title: "Pick one", Description: "Select description.", Default: "a",
		Options: []Option{
			{Label: "Short", Value: "a", Desc: "x"},
			{Label: "Medium (Recommended)", Value: "b", Desc: "a much longer description string"},
		}}
	return qa, qb, qs
}

func TestProbeT586FeasibilityButtonLeft(t *testing.T) {
	locale := "ko"
	result := &WizardResult{}
	qa, qb, _ := t586Questions()
	form := huh.NewForm(huh.NewGroup(
		buildConfirmField(qa, result, &locale).WithButtonAlignment(lipgloss.Left),
		buildConfirmField(qb, result, &locale).WithButtonAlignment(lipgloss.Left),
	)).WithTheme(newMoAIWizardTheme()).WithAccessible(false)
	t586Dump(t, "feas-button-left", newFormDriver(t, form).view())
}

func TestProbeT586FeasibilityFieldSeparator(t *testing.T) {
	locale := "ko"
	result := &WizardResult{}
	qa, qb, _ := t586Questions()
	styles := moaiWizardStyles(false)
	styles.FieldSeparator = lipgloss.NewStyle().SetString("\n")
	form := huh.NewForm(huh.NewGroup(
		buildConfirmField(qa, result, &locale),
		buildConfirmField(qb, result, &locale),
	)).WithTheme(huh.ThemeFunc(func(bool) *huh.Styles { return styles })).WithAccessible(false)
	t586Dump(t, "feas-separator-single", newFormDriver(t, form).view())
}

func TestProbeT586FeasibilityKeyMapHelp(t *testing.T) {
	locale := "ko"
	result := &WizardResult{}
	qa, _, _ := t586Questions()
	km := huh.NewDefaultKeyMap()
	km.Confirm.Toggle.SetHelp("←/→", "전환")
	km.Confirm.Next.SetHelp("enter", "다음")
	form := huh.NewForm(huh.NewGroup(
		buildConfirmField(qa, result, &locale),
	)).WithTheme(newMoAIWizardTheme()).WithKeyMap(km).WithAccessible(false)
	t586Dump(t, "feas-keymap-help", newFormDriver(t, form).view())
}
