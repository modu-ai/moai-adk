FC = "internal/cli/factory_card.go"


def emit(name, edits):
    d = os.path.join(S, name)
    os.makedirs(d, exist_ok=True)
    rep = {}
    for rel, txt in edits.items():
        out = os.path.join(d, rel.replace("/", "__"))
        open(out, "w").write(txt)
        rep[os.path.join(R, rel)] = out
    json.dump({"Replace": rep}, open(os.path.join(d, "overlay.json"), "w"), indent=1)


def sub(text, old, new, count=1):
    n = text.count(old)
    assert n == count, "expected %d of %r, found %d" % (count, old[:90], n)
    return text.replace(old, new)

    ind, arm, cid, call = m.groups()
    return ('%sreturn mu2Defer(ctx, func() (homestate.Card, bool, bool, error) {\n%s\tif err := factoryLeaseBeforeClaim("%s", %s); err != nil {\n%s\t\treturn homestate.Card{}, false, false, err\n%s\t}\n%s\t%s\n%s})\n'
            % (ind, ind, arm, cid, ind, ind, ind, call, ind))
t2, n = pat.subn(repl, t)
