def read(p):
    return open(os.path.join(R, p)).read()


def sub(text, old, new, count=1):
    n = text.count(old)
    assert n == count, "expected %d occurrence(s) of %r, found %d" % (count, old[:90], n)
    return text.replace(old, new)


def emit(name, edits, extra=None):
    d = os.path.join(S, name)
    os.makedirs(d, exist_ok=True)
    rep = {}
    for rel, txt in edits.items():
        out = os.path.join(d, rel.replace("/", "__"))
        open(out, "w").write(txt)
        rep[os.path.join(R, rel)] = out
    for rel, txt in (extra or {}).items():
        out = os.path.join(d, rel.replace("/", "__"))
        open(out, "w").write(txt)
        rep[os.path.join(R, rel)] = out
    json.dump({"Replace": rep}, open(os.path.join(d, "overlay.json"), "w"), indent=1)
